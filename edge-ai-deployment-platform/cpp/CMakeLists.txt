cmake_minimum_required(VERSION 3.20)
project(edgeai_native LANGUAGES CXX)
option(EDGEAI_ENABLE_SANITIZERS "Enable ASan/UBSan" OFF)
add_library(edgeai_native src/int8_linear.cpp)
target_include_directories(edgeai_native PUBLIC include)
target_compile_features(edgeai_native PUBLIC cxx_std_20)
target_compile_options(edgeai_native PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(EDGEAI_ENABLE_SANITIZERS)
  target_compile_options(edgeai_native PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(edgeai_native PRIVATE -fsanitize=address,undefined)
endif()
add_executable(edgeai_native_tests tests/test_main.cpp)
target_link_libraries(edgeai_native_tests PRIVATE edgeai_native)
target_compile_options(edgeai_native_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(EDGEAI_ENABLE_SANITIZERS)
  target_compile_options(edgeai_native_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(edgeai_native_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME edgeai_native_tests COMMAND edgeai_native_tests)
